<script>
(function () {
  var BTN = "9d4e9915fbae403ab44a5b7bb6dba540";
  var PERIOD = 30 * 60 * 1000;

  function now() { return new Date().toLocaleTimeString(); }

  function fire() {
    var t = document.getElementById("tick");
    var began = now();
    if (t) { t.innerText = "▶ 시작 " + began + " · 실행 중…"; }

    // click 은 동기 실행이라 그 줄에서 멈춘다.
    // 위 문구가 화면에 반영된 뒤 누르도록 한 틱 미룬다.
    setTimeout(function () {
      var b = document.getElementById(BTN);
      if (b) { b.click(); }
      if (t) { t.innerText = "▶ 시작 " + began + " · 완료 " + now(); }
    }, 50);
  }

  setTimeout(fire, 5000);        // 파일 열고 5초 뒤 1회
  setInterval(fire, PERIOD);     // 이후 30분마다
})();
</script>
