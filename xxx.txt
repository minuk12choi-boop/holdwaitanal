$roots = @("C:\Program Files\TIBCO", "C:\Program Files (x86)\TIBCO",
           "$env:LOCALAPPDATA\TIBCO", "$env:APPDATA\TIBCO")
foreach ($r in $roots) {
  if (Test-Path $r) {
    Write-Host "=== $r" -ForegroundColor Cyan
    Get-ChildItem $r -Recurse -Filter python.exe -ErrorAction SilentlyContinue |
      Select-Object -ExpandProperty FullName
  } else {
    Write-Host "없음: $r" -ForegroundColor DarkGray
  }
}




Get-ChildItem C:\ -Directory -ErrorAction SilentlyContinue |
  Where-Object { $_.Name -match "TIBCO|Spotfire" } | Select-Object FullName
Get-ChildItem "$env:LOCALAPPDATA","$env:APPDATA" -Directory -ErrorAction SilentlyContinue |
  Where-Object { $_.Name -match "TIBCO|Spotfire" } | Select-Object FullName
