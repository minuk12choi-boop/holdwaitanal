<SpotfireControl id="ed287e84b3004b02927b7f05846183f6" />

<script>
(function () {
  var BTN = "ed287e84b3004b02927b7f05846183f6";
  var PERIOD = 30 * 60 * 1000;

  function now() { return new Date().toLocaleTimeString(); }

  function fire() {
    var t = document.getElementById("tick");
    var began = now();
    if (t) { t.innerText = "시작 " + began + " · 실행 중…"; }

    // click 이 동기 실행이라 그 줄에서 멈춘다.
    // 위 문구가 화면에 반영된 뒤 누르도록 한 틱 미룬다.
    setTimeout(function () {
      var b = document.getElementById(BTN);
      if (b) { b.click(); }
      if (t) { t.innerText = "시작 " + began + " · 완료 " + now(); }
    }, 50);
  }

  setTimeout(fire, 5000);
  setInterval(fire, PERIOD);
})();
</script>
