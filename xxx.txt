<script>
(function () {
  var PERIOD   = 15 * 60 * 1000;    // 15분 주기
  var MAX_WAIT = 12 * 60 * 1000;    // busy 최대 유지. PERIOD 보다 작아야 한다.

  var busy = false;
  var startedAt = 0;

  function now() { return new Date().toLocaleTimeString(); }

  // 이 텍스트 영역 안의 버튼을 찾는다. id 나 이름에 의존하지 않는다.
  function findBtn() {
    var tick = document.getElementById("tick");
    if (!tick) { return null; }
    var box = tick.parentNode;                 // 텍스트 영역 컨테이너
    for (var i = 0; i < 5 && box; i++) {
      var b = box.querySelectorAll(
        "input[type='button'], input[type='submit'], button");
      if (b.length) { return b[0]; }           // 버튼은 하나뿐
      box = box.parentNode;                    // 없으면 한 단계 위로
    }
    return null;
  }

  function fire() {
    var t = document.getElementById("tick");
    var b = findBtn();

    if (!b) {
      if (t) { t.innerText = "✕ 버튼을 찾지 못함 " + now(); }
      return;
    }
    if (busy && Date.now() - startedAt < MAX_WAIT) {
      if (t) { t.innerText += " · " + now() + " 건너뜀"; }
      return;
    }

    busy = true;
    startedAt = Date.now();
    var began = now();
    if (t) { t.innerText = "▶ 시작 " + began + " · 실행 중…"; }
    setTimeout(function () { b.click(); }, 50);

    var watch = setInterval(function () {
      var log = document.body.innerText || "";
      if (log.indexOf("완료 (테이블") >= 0 && Date.now() - startedAt > 10000) {
        busy = false; clearInterval(watch);
        if (t) { t.innerText = "▶ 시작 " + began + " · 완료 " + now(); }
      } else if (Date.now() - startedAt > MAX_WAIT) {
        busy = false; clearInterval(watch);
        if (t) { t.innerText = "▶ 시작 " + began + " · 시간초과"; }
      }
    }, 5000);
  }

  setTimeout(fire, 60000);
  setInterval(fire, PERIOD);
})();
</script>
