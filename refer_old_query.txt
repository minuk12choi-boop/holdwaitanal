1. step_kfr7
WITH
m AS (                             
  SELECT /*+ MATERIALIZE */ lot_id, order_seq
  FROM   smicdc_nrdk.mc_lot
  WHERE  lot_status_seg IN ('Active','Hold')
),
r AS (                                   
  SELECT lot_id, order_seq,
         COUNT(CASE WHEN delay_step_type = 'S' THEN 1 END)
           OVER (PARTITION BY lot_id ORDER BY order_seq) AS de_rank
  FROM   smicdc_nrdk.mc_lot_step_path
  WHERE  delay_step_type IN ('S','Y')
    AND  lot_id IN (SELECT lot_id FROM m)
),
k AS (
  SELECT
    'KFR7'                          AS line,
    sp.lot_id,
    sp.proc_id,
    sp.order_seq,
    sp.delay_step_type,
    r.de_rank,
    CASE
      WHEN sp.delay_step_type = 'S' THEN '연속첫'
      WHEN sp.delay_step_type = 'Y'
        THEN '연속(' || TO_CHAR(TRUNC(sp.delay_time_mins)) || ')'
    END                             AS 연속,
    sp.layer_id,
    TO_CHAR(sp.step_level)          AS step_level,
    sp.step_seq,
    sp.step_desc,
    sp.eqp_type,
    sp.eqp_group_id                 AS eqp_group_raw,
    sp.recipe_id,
    sp.ext_1st_vals                 AS tkintype,
    sp.tkin_type_detail
  FROM       smicdc_nrdk.mc_lot_step_path sp
  JOIN       m  ON  m.lot_id    = sp.lot_id
              AND  m.order_seq <= sp.order_seq
  LEFT JOIN  r  ON  r.lot_id    = sp.lot_id
              AND  r.order_seq  = sp.order_seq
  WHERE      sp.step_skip_yn <> 'Y'
),
eqp_group AS (
  SELECT DISTINCT line_id, eqp_group_name, eqp_id
  FROM   smimes.mi_eqp_group_list
  WHERE  line_id = 'KFR7'
    AND  INSTR(eqp_id, 'OFF') = 0
),
equipment AS (
  SELECT DISTINCT
         line_id, origin_line_id AS eqpline, batch_kind, eqp_id,
         eqp_status, tool_kind,
         TO_DATE(eqp_status_change_time, 'yyyymmdd hh24:mi:ss') AS eqp_status_change_time
  FROM   smimes.mi_equipment
  WHERE  line_id = 'KFR7'
    AND  tool_kind IN ('EQP','CHAMBER')
)
SELECT
  u.line, u.lot_id, u.proc_id, u.order_seq, u.de_rank, u.delay_step_type,
  u.연속, u.layer_id, u.step_level,
  NVL(CASE WHEN u.tkin_type_detail = '-' THEN NULL ELSE u.tkin_type_detail END,
      u.tkintype)                       AS ein,
  u.step_seq, u.step_desc, u.eqp_type, u.eqp_group_raw, u.recipe_id,
  e.eqp_id,
  ee.batch_kind, ee.eqpline,
  ee.eqp_status                         AS body_status,
  ee.eqp_status_change_time
FROM       k u
LEFT JOIN  eqp_group e
      ON   e.eqp_group_name = u.eqp_group_raw
     AND   e.line_id        = u.line
LEFT JOIN  equipment ee
      ON   ee.line_id = e.line_id
     AND   ee.eqp_id  = e.eqp_id
ORDER BY   u.lot_id, u.order_seq, e.eqp_id

3. tip_kfr7
with

t as (
select DISTINCT
'KFR7' as line,
t.process,
t.step,
t.ppid,
t.eqpid,
'-' as lot_type,
(case when t.chamberid in ('-', 'MAIN') then 'MAIN' else t.chamberid end) as chamberid,
(case 
when t.type = 'DOING' then 'DOING'
when t.type='PREVENT' AND NVL(t.checkcount,0)=0 then 'PREVENT' 
when t.type='PREVENT' AND NVL(t.tkin_count,0)>=NVL(checkcount,0) then 'PREVENT' 
when t.type='PREVENT' AND NVL(t.tkin_count,0)<NVL(checkcount,0) then 'DOING' 
end) as prevent,
(case when t.chamberid is not null and t.chamberid not in ('MAIN', '-') then (t.eqpid||'-'||chamberid) else t.eqpid end) as ee,
nvl(eventtime, updated) as eventtime
from smicdc_nrdk.trackinprevent t
),

ttt as (select * 
from (select
	t.*,
	(count(case when t.prevent = 'DOING' then t.prevent end) over (partition by t.line, t.process, t.step, t.ppid, t.ee, t.lot_type)) as c,
	(count(t.ee) over (partition by t.line, t.process, t.step, t.ppid, t.ee, t.lot_type)) as cc,
	(max(t.eventtime) over (partition by t.line, t.process, t.step, t.ppid, t.ee, t.prevent, t.lot_type)) as m
	from t) tt
where tt.cc>1 and tt.c>0 and tt.prevent='DOING' and tt.m=tt.eventtime
	or tt.cc>1 and tt.c=0 and tt.m=tt.eventtime
	or tt.cc=1),

e as (
select DISTINCT
line_id, 
origin_line_id as eqpline,
batch_kind,
eqp_id, 
eqp_status, 
tool_kind,
TO_DATE(eqp_status_change_time, 'yyyymmdd hh24:mi:ss') as eqp_status_change_time
from smimes.mi_equipment 
where line_id in ('KFR7') and tool_kind in ('EQP', 'CHAMBER')
),

te as (select 
ttt.line,
ttt.process,
ttt.step,
ttt.ppid,
ttt.eqpid,
ttt.chamberid,
ttt.lot_type,
ttt.prevent,
ttt.ee,
ttt.eventtime,
e.batch_kind,
e.eqpline
from ttt, e
where ttt.ee=e.eqp_id(+)
	and ttt.line=e.line_id(+)
order by process, step, ppid, ee),

tee as (select
a.line,
a.process,
a.step,
a.ppid,
a.eqpid,
NVL(case when NVL(a.batch_kind, '-') in ('BATCH_FURNACE', 'BATCH_WET') then NULL else b.chamberid end, NULL) as chamberid,
NVL(case when NVL(a.batch_kind, '-') in ('BATCH_FURNACE', 'BATCH_WET') then a.eqpid else b.ee end, a.eqpid) as eqpcham,
nvl(b.lot_type, a.lot_type) as lot_type,
a.batch_kind,
(case when a.prevent='PREVENT' or b.prevent='PREVENT' then 'PREVENT' else 'DOING' end) as prevent,   
a.prevent as type_body,
b.prevent as type_cham,          
NVL(case when NVL(a.batch_kind, '-') in ('BATCH_FURNACE', 'BATCH_WET') then a.eventtime else b.eventtime end, a.eventtime) as eventtime,
NVL(a.eqpline, b.eqpline) as eqpline
from (select * from te where NVL(chamberid,'-') in ('MAIN', '-')) a,
       (select * from te where NVL(chamberid,'-') not in ('MAIN', '-')) b
where a.line=b.line(+)
	and a.process=b.process(+)
	and a.step=b.step(+)
	and a.ppid=b.ppid(+)
	and a.eqpid=b.eqpid(+)),

es as (select DISTINCT
a.line_id,
nvl(b.eqp_id, a.eqp_id) as eqpcham,
a.eqp_status as body_eqp_status,
b.eqp_status as cham_eqp_status,
a.eqp_status_change_time as body_status_change_time,
b.eqp_status_change_time as cham_status_change_time
FROM (select line_id, eqp_id, eqp_status, eqp_status_change_time from e where  tool_kind = 'EQP') a,
         (select line_id, nvl(SUBSTR(EQP_ID, 1, INSTR(EQP_ID, '-')-1),SUBSTR(EQP_ID, 1, INSTR(EQP_ID, '_')-1)) AS body, eqp_id, eqp_status, eqp_status_change_time from e where tool_kind = 'CHAMBER') b
where a.eqp_id = b.body(+)
	and a.line_id=b.line_id(+))

select DISTINCT
line,
process,
step,
ppid,
tee.eqpid,
tee.eqpcham,
chamberid,
'-' as lot_type,
batch_kind,
prevent,
type_body,
type_cham,
(case when prevent='PREVENT' then eventtime end) as tip_eventtime,
(case when body_eqp_status in ('LOCAL', 'PM', 'DOWN') then body_eqp_status 
       when cham_eqp_status in ('LOCAL' ,'PM', 'DOWN') then cham_eqp_status end) as eqpissue,
body_eqp_status,
cham_eqp_status,
(case when body_eqp_status in ('LOCAL', 'PM', 'DOWN') then body_status_change_time
       when cham_eqp_status in ('LOCAL' ,'PM', 'DOWN') then cham_status_change_time end) as eqpissuetime,
eqpline
 from tee, es
where tee.eqpcham = es.eqpcham(+)
	and tee.line = es.line_id(+)

5. mc_lot
WITH

m as (
select
'PFR1' as line,
sys_line_id,
cur_line_id,
origin_line_id,
lot_id,
last_event_date
from smicdc_p3nrd.mc_lot

union all

select
'KFR7' as line,
sys_line_id,
cur_line_id,
origin_line_id,
lot_id,
last_event_date
from smicdc_nrdk.mc_lot
),

c as (
select
distinct
lot_id,
max(last_event_date) over (partition by lot_id) as m
FROM m),

m0 as (select 
m.line, 
m.lot_id,
c.m as max_event_date
from m, c
where m.lot_id = c.lot_id(+)
	and m.last_event_date = c.m),


T1 AS	
(SELECT LINE_ID, LOT_ID, NEW_ATTR_VALUE
FROM 
(SELECT line_id, LOT_ID, NEW_ATTR_VALUE, MAX(LOT_TRANSN_TIME) OVER (PARTITION BY LOT_ID,  LINE_ID) AS M, SUM(CASE WHEN WIP_ATTRIBUTE='FLOWLEVEL' THEN 1 END) OVER (PARTITION BY LOT_ID, STEP_SEQ, LINE_ID) SUM, LOT_TRANSN_TIME
	
FROM   MI_LOT_TRANSN_HIST_V	
	
WHERE LOT_TRANSN_TYPE = 'ModifyAttr' AND WIP_ATTRIBUTE IN ('GRADE')
	AND LINE_ID IN ('PFR1', 'KFR7'))

WHERE SUM IS NULL AND M=LOT_TRANSN_TIME),
	
g AS (SELECT DISTINCT T1.LINE_ID, T1.LOT_ID, T1.NEW_ATTR_VALUE AS GRADE
FROM T1 
ORDER BY LOT_ID),


m1 as ((select
'PFR1' as line,
m.cur_line_id,
m.sys_line_id,
m.origin_line_id,
m.lot_id,
m.carr_id,
m.lot_type,
(case when nvl(g.grade,'-')<>'-' then 'G'||g.grade end) as grade,
TO_CHAR(m.lot_level) as lot_level,
m.cur_qty,
m.bay_name,
(case when m.lot_status_seg='Hold' then 'HOLD' else m.step_status_seg end) as status,
m.proc_id,
TO_CHAR(m.order_seq) as order_seq,
m.step_seq,
to_date(m.START_DATE, 'yyyymmdd hh24:mi:ss') as START_DATE,
to_date(m.LAST_TKOUT_DATE, 'yyyymmdd hh24:mi:ss') as LAST_TKOUT_DATE,
to_date(m.STEP_ARRIVE_DATE, 'yyyymmdd hh24:mi:ss') as STEP_ARRIVE_DATE,
to_date(m.LAST_EVENT_DATE, 'yyyymmdd hh24:mi:ss') as LAST_EVENT_DATE

from smicdc_p3nrd.mc_lot m, m0, g
where m.last_event_date = m0.max_event_date
	and m.lot_id = m0.lot_id
	and lot_status_seg in ('Active', 'Hold')
	and m.order_seq is not null
	and m.lot_id = g.lot_id(+)
	and 'PFR1' = g.line_id(+))

union all

(select
'KFR7' as line,
m.cur_line_id,
m.sys_line_id,
m.origin_line_id,
m.lot_id,
m.carr_id,
m.lot_type,
(case when nvl(g.grade,'-')<>'-' then 'G'||g.grade end) as grade,
TO_CHAR(m.lot_level) as lot_level,
m.cur_qty,
m.bay_name,
(case when m.lot_status_seg='Hold' then 'HOLD' else m.step_status_seg end) as status,
m.proc_id,
TO_CHAR(m.order_seq) as order_seq,
m.step_seq,
to_date(m.START_DATE, 'yyyymmdd hh24:mi:ss') as START_DATE,
to_date(m.LAST_TKOUT_DATE, 'yyyymmdd hh24:mi:ss') as LAST_TKOUT_DATE,
to_date(m.STEP_ARRIVE_DATE, 'yyyymmdd hh24:mi:ss') as STEP_ARRIVE_DATE,
to_date(m.LAST_EVENT_DATE, 'yyyymmdd hh24:mi:ss') as LAST_EVENT_DATE

from smicdc_nrdk.mc_lot m, m0, g
where m.last_event_date = m0.max_event_date
	and m.lot_id = m0.lot_id
	and lot_status_seg in ('Active', 'Hold')
	and m.lot_id = g.lot_id(+)
	and 'KFR7' = g.line_id(+))
),

co as (
select
*
from (select 
'PFR1' as line,
lot_id, 
step_comment as lot_inform,
update_date,
max(update_date) over (partition by lot_id) as m
from SMICDC_P3NRD.MC_LOT_STEP_COMMENT 
where parent_order_seq=0
	and comment_type = 'LOT')
where m=update_date

union all

select
*
from (select 
'KFR7' as line,
lot_id, 
step_comment as lot_inform,
update_date,
max(update_date) over (partition by lot_id) as m
from SMICDC_NRDK.MC_LOT_STEP_COMMENT 
where parent_order_seq=0
	and comment_type = 'LOT')
where m=update_date)



select 
distinct 
m1.* ,
(case 
when origin_line_id<>sys_line_id then 'OLD_SEND'
when origin_line_id=sys_line_id and sys_line_id<>cur_line_id then 'NEW_SEND'
end) as sendfab,
co.lot_inform
from m1, co
where m1.line=m1.sys_line_id

	and cur_line_id not in ('CHTV')
	and m1.line = co.line(+)
	and m1.lot_id = co.lot_id(+)

6. hold
with

hh as (select
distinct
line_id,
item_type,
lot_id,
step_seq,
hold_user,
hold_reason,
hold_date,
max(hold_date) over (partition by line_id, lot_id, step_seq, item_type) as m
from (SELECT
version_desc,
max(version_desc) over (partition by line_id, version_desc) as m,
line_id,
item_type,
lot_id,
step_seq,
hold_user_name as hold_user,
issue_reason_cont as hold_reason,
issue_date as hold_date
FROM MEMMSS.FAB_ISSUE_LOT 
WHERE line_id in ('PFR1', 'KFR7')
	and status_seq <> '2') h
where m=version_desc),

h0 as (select
rownum as r,
line_id,
item_type,
lot_id,
step_seq,
hold_user,
hold_reason,
hold_date
from (select * from hh order by line_id, item_type, lot_id, step_seq) hh
where m=hold_date
order by line_id, item_type, lot_id, step_seq)


select
h0.*
from h0, (select distinct line_id, lot_id ,step_seq, max(r) over (partition by line_id, lot_id, step_seq) as m from h0 where item_type = 'EXCEPTION') h1
where h0.r = h1.m

union all
