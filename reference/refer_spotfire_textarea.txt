<SCRIPT>
(function () {
  var BTN_ID      = "0963b0fd531a4c3caaf159001748e373";  // RefreshAll 버튼
  var PERIOD_MIN  = 30;                 // 벽시계 :00, :30 에 실행
  var MIN_GAP_MIN = 15;                 // 직전 완료 후 이만큼 안 지났으면 건너뜀
  var FIRST_DELAY = 60000;              // 문서를 연 뒤 첫 실행까지
  var MAX_WAIT    = 15 * 60 * 1000;     // 이 시간 넘으면 멈춘 것으로 보고 해제

  // 상태를 window 에 둬야 텍스트영역이 다시 그려져도 중복 실행을 막을 수 있다.
  //   off  : 사용자가 잠시 꺼 둔 상태. 문서를 다시 열면 켜진 상태로 돌아온다.
  var S = window.__MFM = window.__MFM || { to: [], iv: [], busy: false,
                                           startedAt: 0, lastEnd: 0,
                                           off: false };
  S.to.forEach(function (t) { clearTimeout(t); });
  S.iv.forEach(function (t) { clearInterval(t); });
  S.to = []; S.iv = [];

  function later(fn, ms) { S.to.push(setTimeout(fn, ms)); }
  function every(fn, ms) { var id = setInterval(fn, ms); S.iv.push(id); return id; }
  function now() { return new Date().toLocaleTimeString(); }
  function say(msg) {
    var t = document.getElementById("tick");
    if (t) { t.innerText = msg; }
  }

  // ── ON / OFF 버튼 ───────────────────────────────────────────────
  //   OFF 를 누르면 예약을 모두 지운다. ON 을 누르면 다시 예약한다.
  //   Spotfire 텍스트영역 안에서만 도는 스크립트라 문서를 닫으면 어차피 멈춘다.
  function paint() {
    var box = document.getElementById("pw");
    if (!box) { return; }
    box.innerHTML =
      '<button id="mfmOn"  type="button">ON</button>' +
      '<button id="mfmOff" type="button">OFF</button>' +
      '<span id="mfmSt" style="margin-left:6px"></span>';

    var on  = document.getElementById("mfmOn");
    var off = document.getElementById("mfmOff");
    var st  = document.getElementById("mfmSt");

    function skin() {
      var base = "font-size:11px;font-weight:700;border-radius:3px;" +
                 "padding:1px 8px;margin-right:4px;cursor:pointer;";
      on.style  = base + (S.off ? "border:1px solid #C9CDD4;background:#fff;color:#6B7280"
                                : "border:1px solid #1F7A46;background:#1F7A46;color:#fff");
      off.style = base + (S.off ? "border:1px solid #B91C1C;background:#B91C1C;color:#fff"
                                : "border:1px solid #C9CDD4;background:#fff;color:#6B7280");
      st.innerText = S.off ? "자동 실행 꺼짐" : "자동 실행 켜짐 (" + PERIOD_MIN + "분)";
      st.style = "font-size:11px;color:" + (S.off ? "#B91C1C" : "#1F7A46");
    }

    on.onclick = function () {
      if (!S.off) { return; }
      S.off = false;
      skin();
      say("· " + now() + " 자동 실행 다시 켬");
      scheduleNext();
    };
    off.onclick = function () {
      if (S.off) { return; }
      S.off = true;
      S.to.forEach(function (t) { clearTimeout(t); });
      S.iv.forEach(function (t) { clearInterval(t); });
      S.to = []; S.iv = []; S.busy = false;
      skin();
      say("■ " + now() + " 자동 실행 멈춤. ON 을 누르면 다시 시작한다.");
    };
    skin();
  }

  // runlog 는 "HH:MM:SS 완료 (테이블 N개)" 형태다.
  // 문구 존재 여부만 보면 '이전 회차 기록' 을 보고 끝났다고 착각한다.
  // 앞의 시각 9글자까지 함께 읽어 값이 '바뀌었는지' 로 판정한다.
  function doneStamp() {
    var log = document.body.innerText || "";
    var i = log.indexOf("완료 (테이블");
    return i >= 9 ? log.substr(i - 9, 9) : "";
  }

  // 버튼은 id 로 먼저 찾고, 편집으로 id 가 바뀌었으면 이 영역 안에서 찾는다.
  function findBtn() {
    var b = document.getElementById(BTN_ID);
    if (b) { return b; }
    var tick = document.getElementById("tick");
    var box = tick ? tick.parentNode : null;
    for (var i = 0; i < 5 && box; i++) {
      var f = box.querySelectorAll(
        "input[type='button'], input[type='submit'], button");
      // ON / OFF 는 우리가 만든 것이라 건너뛴다.
      for (var k = 0; k < f.length; k++) {
        if (f[k].id !== "mfmOn" && f[k].id !== "mfmOff") { return f[k]; }
      }
      box = box.parentNode;
    }
    return null;
  }

  function fire(reason) {
    if (S.off) { return; }             // 꺼 둔 상태면 아무것도 하지 않는다
    var b = findBtn();
    if (!b) { say("✕ 버튼을 찾지 못함 " + now()); return; }

    if (S.busy && Date.now() - S.startedAt < MAX_WAIT) {
      say("· " + now() + " 건너뜀(실행 중)"); return;
    }
    if (MIN_GAP_MIN && S.lastEnd &&
        Date.now() - S.lastEnd < MIN_GAP_MIN * 60 * 1000) {
      var left = Math.ceil(
        (MIN_GAP_MIN * 60 * 1000 - (Date.now() - S.lastEnd)) / 60000);
      say("· " + now() + " 건너뜀(" + left + "분 더)"); return;
    }

    S.busy = true;
    S.startedAt = Date.now();
    var began = now();
    var mark = doneStamp();            // 누르기 직전의 runlog 시각
    say("▶ 시작 " + began + " (" + reason + ") · 실행 중…");
    later(function () { b.click(); }, 50);

    var watch = every(function () {
      var cur = doneStamp();
      if (cur && cur !== mark && Date.now() - S.startedAt > 10000) {
        S.busy = false; S.lastEnd = Date.now(); clearInterval(watch);
        say("▶ 시작 " + began + " · 완료 " + now());
      } else if (Date.now() - S.startedAt > MAX_WAIT) {
        S.busy = false; S.lastEnd = Date.now(); clearInterval(watch);
        say("▶ 시작 " + began + " · 시간초과");
      }
    }, 5000);
  }

  function msToNextSlot() {
    var d = new Date();
    var next = (Math.floor(d.getMinutes() / PERIOD_MIN) + 1) * PERIOD_MIN;
    var t = new Date(d);
    t.setMinutes(next, 0, 0);
    return t - d;
  }
  function scheduleNext() {
    if (S.off) { return; }
    later(function () { fire("정기"); scheduleNext(); }, msToNextSlot());
  }

  paint();
  if (S.off) {
    say("■ 자동 실행 꺼짐. ON 을 누르면 다시 시작한다.");
    return;
  }
  if (!S.armed) { S.armed = true; later(function () { fire("최초"); }, FIRST_DELAY); }
  scheduleNext();
})();
</SCRIPT>
