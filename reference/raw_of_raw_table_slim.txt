-- raw_of_raw_table_slim.txt
--   메모리 부족(MemoryError)을 피하려고 **표를 여섯으로 쪼개는** 쿼리다.
--   기존 raw_of_raw_table.txt 는 그대로 두고, 이 두 개만 갈아 끼운다.
--
--   [왜 조건절로 거르지 않는가]
--     process 로 걸러 봤더니 100만 행 줄이는 데 호출이 6분 늘었다.
--     PK 가 아닌 컬럼이라 전체를 훑기 때문이다. 행을 고르게 3등분해
--     각 데이터 함수가 1/3 씩만 읽게 하는 편이 빠르고 확실하다.
--
--   [왜 셋인가]
--     Spotfire 는 스크립트를 돌리기 전에 표 전체를 메모리로 읽는다.
--     한 함수가 700만 행을 읽다 죽으므로, 120만 행씩 여섯 번 읽게 한다.
--
--   [나누는 기준]
--     ORA_HASH(eqpid, 5) 를 쓴다. 0 ~ 5 여섯 조각이 된다.
--       - 같은 설비는 늘 같은 조각에 간다(회차마다 배치가 바뀌지 않는다)
--       - eqpid 분포가 고르므로 여섯 조각 크기도 비슷하다
--       - 인덱스와 무관해 조회가 느려지지 않는다
--     ROWNUM 은 쓰지 않는다. 정렬 전에 붙는 값이라 회차마다 달라진다.
--
--   [읽는 쪽]
--     build_f3 는 세 조각을 이어 붙여 한 표로 다룬다. 합치면 원본과 같다.
--------------------------------------------------------------------------------


--------------------------------------------------------------------------------
-- 4) PFR1_KFR7_TIP_0  /  _1  /  _2
--
--    아래 쿼리를 여섯 벌 만든다. 맨 아래 두 줄의 숫자만 0 · 1 · 2 로 바꾼다.
--
--      표 이름 : PFR1_KFR7_TIP_0 · PFR1_KFR7_TIP_1 · PFR1_KFR7_TIP_2
--      함수    : s3drive_tip_0 ~ s3drive_tip_5
--                (입력은 그 표 하나씩만 등록한다)
--
--    lot_type 컬럼은 없다. python 이 '-'(와일드카드)로 채운다.
--------------------------------------------------------------------------------
SELECT sysdate as query_time,
'PFR1' as line_id,
       process,
       step,
       ppid,
       eqpid,
       chamberid,
       type,
       checkcount,
       tkin_count,
       updated,
       eventtime
FROM   smicdc_p3nrd.trackinprevent
WHERE  owner IN ('LEVEL1', 'PHOTO_LEVEL1')
  AND  ORA_HASH(eqpid, 5) = 0          -- <<< 0 ~ 5 로 바꿔 여섯 벌

union all

SELECT sysdate as query_time,
'KFR7' as line_id,
       process,
       step,
       ppid,
       eqpid,
       chamberid,
       type,
       checkcount,
       tkin_count,
       updated,
       eventtime
FROM   smicdc_nrdk.trackinprevent
WHERE  owner IN ('LEVEL1', 'PHOTO_LEVEL1')
  AND  ORA_HASH(eqpid, 5) = 0          -- <<< 위와 같은 숫자로 맞춘다
;
