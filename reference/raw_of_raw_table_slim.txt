-- 4) PFR1_KFR7_TIP  (행을 줄인 판)
--
--    재공 lot 이 실제로 지나는 공정(proc_id)의 규칙만 가져온다.
--    **와일드카드 '-' 는 반드시 남긴다.** 그것이 없으면 특정 공정을
--    지정하지 않은 전역 규칙이 통째로 사라진다.
--
--    lot_type 컬럼은 없다. python 이 '-'(와일드카드)로 채운다.
--------------------------------------------------------------------------------
SELECT sysdate as query_time,
'PFR1' as line_id,
       process,
       step,
       ppid,
       eqpid,
       chamberid,
       type,
       checkcount,
       tkin_count,
       updated,
       eventtime
FROM   smicdc_p3nrd.trackinprevent t
WHERE  owner IN ('LEVEL1', 'PHOTO_LEVEL1')
  AND (t.process IS NULL
       OR t.process = '-'
       OR t.process IN (SELECT DISTINCT proc_id
                        FROM   smicdc_p3nrd.mc_lot
                        WHERE  lot_status_seg IN ('Active', 'Hold')))

union all

SELECT sysdate as query_time,
'KFR7' as line_id,
       process,
       step,
       ppid,
       eqpid,
       chamberid,
       type,
       checkcount,
       tkin_count,
       updated,
       eventtime
FROM   smicdc_nrdk.trackinprevent t
WHERE  owner IN ('LEVEL1', 'PHOTO_LEVEL1')
  AND (t.process IS NULL
       OR t.process = '-'
       OR t.process IN (SELECT DISTINCT proc_id
                        FROM   smicdc_nrdk.mc_lot
                        WHERE  lot_status_seg IN ('Active', 'Hold')))
;
