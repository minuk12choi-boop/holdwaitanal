WITH
m AS (
  SELECT /*+ MATERIALIZE */ lot_id, order_seq
  FROM   smicdc_p3nrd.mc_lot
  WHERE  lot_status_seg IN ('Active','Hold')
),
r AS (
  SELECT lot_id, order_seq,
         COUNT(CASE WHEN delay_step_type = 'S' THEN 1 END)
           OVER (PARTITION BY lot_id ORDER BY order_seq) AS de_rank
  FROM   smicdc_p3nrd.mc_lot_step_path
  WHERE  delay_step_type IN ('S','Y')
    AND  lot_id IN (SELECT lot_id FROM m)
),
p AS (
  SELECT
    'PFR1'                          AS line,
    sp.lot_id,
    sp.proc_id,
    sp.order_seq,
    sp.delay_step_type,
    r.de_rank,
    CASE
      WHEN sp.delay_step_type = 'S' THEN '연속첫'
      WHEN sp.delay_step_type = 'Y'
        THEN '연속(' || TO_CHAR(TRUNC(sp.delay_time_mins)) || ')'
    END                             AS 연속,
    sp.layer_id,
    TO_CHAR(sp.step_level)          AS step_level,
    sp.step_seq,
    sp.step_desc,
    sp.eqp_type,
    sp.eqp_group_id                 AS eqp_group_raw,
    sp.recipe_id,
    sp.ext_1st_vals                 AS tkintype,
    sp.tkin_type_detail
  FROM       smicdc_p3nrd.mc_lot_step_path sp
  JOIN       m  ON  m.lot_id    = sp.lot_id
              AND  m.order_seq <= sp.order_seq
  LEFT JOIN  r  ON  r.lot_id    = sp.lot_id
              AND  r.order_seq  = sp.order_seq
  WHERE      sp.step_skip_yn <> 'Y'
),
eqp_group AS (
  SELECT DISTINCT line_id, eqp_group_name, eqp_id
  FROM   smimes.mi_eqp_group_list
  WHERE  line_id = 'PFR1'
    AND  INSTR(eqp_id, 'OFF') = 0
),
equipment AS (
  SELECT DISTINCT
         line_id, origin_line_id AS eqpline, batch_kind, eqp_id,
         eqp_status, tool_kind,
         TO_DATE(eqp_status_change_time, 'yyyymmdd hh24:mi:ss') AS eqp_status_change_time
  FROM   smimes.mi_equipment
  WHERE  line_id = 'PFR1'
    AND  tool_kind IN ('EQP','CHAMBER')
)
SELECT
  u.line, u.lot_id, u.proc_id, u.order_seq, u.de_rank, u.delay_step_type,
  u.연속, u.layer_id, u.step_level,
  NVL(CASE WHEN u.tkin_type_detail = '-' THEN NULL ELSE u.tkin_type_detail END,
      u.tkintype)                       AS ein,
  u.step_seq, u.step_desc, u.eqp_type, u.eqp_group_raw, u.recipe_id,
  e.eqp_id,
  ee.batch_kind, ee.eqpline,
  ee.eqp_status                         AS body_status,
  ee.eqp_status_change_time
FROM       p u
LEFT JOIN  eqp_group e
      ON   e.eqp_group_name = u.eqp_group_raw
     AND   e.line_id        = u.line
LEFT JOIN  equipment ee
      ON   ee.line_id = e.line_id
     AND   ee.eqp_id  = e.eqp_id
ORDER BY   u.lot_id, u.order_seq, e.eqp_id
