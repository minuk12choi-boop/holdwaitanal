WITH

S AS (select  DISTINCT ORIGIN_ATTR_VALUE, NEW_ATTR_VALUE, TO_DATE(SUBSTR(LOT_TRANSN_TIME,1,15), 'YYYYMMDD HH24:MI:SS') AS SPLIT_DATE

from SMIMES.MI_LOT_TRANSN_HIST

WHERE LINE_ID IN ('PFR1')
	AND LOT_TRANSN_TYPE='Split' 
	AND WIP_ATTRIBUTE IN ('LOTID')
	AND SUBSTR(ORIGIN_ATTR_VALUE,1,1)='1'
	AND INSTR(NEW_ATTR_VALUE, '.')>INSTR(NEW_ATTR_VALUE, 'E0', -1)),

MC AS (SELECT 
(CASE WHEN LOT_STATUS_SEG='Finished' OR LOT_STATUS_SEG='Archivable' THEN 'F/O' ELSE '중도F/O' END) AS 구분1, LOT_ID, LOT_TYPE, START_QTY, 

(CASE WHEN LOT_STATUS_SEG='Terminated' THEN 0 ELSE CUR_QTY END) AS FO매수, 


TO_DATE(START_DATE, 'YYYYMMDD HH24:MI:SS') AS START_DATE, TO_DATE(LAST_EVENT_DATE, 'YYYYMMDD HH24:MI:SS') AS LAST_EVENT_DATE,
(TO_DATE(LAST_EVENT_DATE, 'YYYYMMDD HH24:MI:SS')-TO_DATE(START_DATE, 'YYYYMMDD HH24:MI:SS')) AS TAT, S.SPLIT_DATE, 

(CASE WHEN SUBSTR(LOT_ID, 1,1) IN ('Q') THEN SUBSTR(LOT_ID,1,1)
         WHEN  SUBSTR(LOT_ID, 1,2) IN ('1S') THEN SUBSTR(LOT_ID,1,2) 
         ELSE SUBSTR(LOT_ID,1,4)||PROC_ID END) AS ID구분1, 
SUBSTR(LOT_ID,1,4) AS ID구분2 , LOT_LEVEL

FROM SMICDC_P3NRD.MC_LOT, S

WHERE LOT_STATUS_SEG IN ('Finished', 'Archivable', 'Terminated')
	AND (ORIGIN_LINE_ID IN ('PFR1'))
	AND INSTR(LOT_ID, '.')>INSTR(LOT_ID, 'E0',-01)
	AND SUBSTR(LOT_ID,1,2) NOT IN ('1I')
	AND SUBSTR(LOT_ID,1,1) NOT IN ('Q', 'P', 'E', 'B')
	AND SUBSTR(LOT_ID,1,4) NOT IN ('1GZZ')
	AND LOT_ID=NEW_ATTR_VALUE(+)
	
),

T AS (select DISTINCT 
(CASE  
WHEN H.EQP_TYPE IN ('FSORTE', 'JSORTE') THEN '소터' 
WHEN UPPER(SUBSTR(H.LOT_TRANSN_DESC,1,1))='S' THEN '샘플' 
WHEN (H.EQP_TYPE IN ('POVLAY')) OR (UPPER(SUBSTR(H.PROCESS_EQP_ID,1,1))='M') THEN '계측'
ELSE '메인'
END) AS 구분2,
CASE WHEN (INSTR(H.LOT_ID, '.1E')>0 OR INSTR(H.LOT_ID, '.2E')>0 OR INSTR(H.LOT_ID, '.3E')>0 OR INSTR(H.LOT_ID, '.4E')>0 OR INSTR(H.LOT_ID, '.5E')>0 OR INSTR(H.LOT_ID, '.6E')>0 OR INSTR(H.LOT_ID, '.7E')>0) THEN SUBSTR(H.LOT_ID, 1, INSTR(H.LOT_ID, '.')+1) ELSE H.LOT_ID END AS REV_LOT_ID , H.LOT_ID, H.LOT_TYPE, H.COMPONENT_QTY, H.REAL_COMPONENT_QTY, H.PROCESS_ID, H.EQP_TYPE, H.PROCESS_EQP_ID, H.CHAMBER_ID, H.DELAY_TIME, H.REWORK_MAIN_STEP, H.REWORK_YN, H.LAYER_ID, H.STEP_SEQ, H.LOT_TRANSN_DESC, H.TKIN_OPERTR_ID, H.LINE_ID, H.CURRENT_LINE_ID, TO_DATE(H.RECENT_TKOUT_TIME, 'YYYYMMDD HH24:MI:SS') AS RECENT_TKOUT_DATE, H.TKIN_DATE, TO_DATE(SUBSTR(H.LOT_TRANSN_TIME,1,15), 'YYYYMMDD HH24:MI:SS') AS TKOUT_DATE, PORT_LOAD_DATE, PORT_UNLOAD_DATE, PROCESS_START_DATE, PROCESS_FINISH_DATE,

(TO_DATE(SUBSTR(H.LOT_TRANSN_TIME,1,15), 'YYYYMMDD HH24:MI:SS') - H.TKIN_DATE) AS NET,
(H.TKIN_DATE - TO_DATE(H.RECENT_TKOUT_TIME, 'YYYYMMDD HH24:MI:SS')) AS WAIT, H.PPID

from SMIMES.MI_LOT_TRANSN_HIST H, MC

WHERE H.LINE_ID IN ('PFR1') AND H.LOT_TRANSN_TYPE='TrackOut' AND MC.LOT_ID=H.LOT_ID AND SUBSTR(H.LOT_ID,1,1) NOT IN ('Q','P','E', 'B'))

SELECT 
MC.구분1, T.구분2, T.REV_LOT_ID, T.LOT_ID, T.LOT_TYPE, T.COMPONENT_QTY, T.REAL_COMPONENT_QTY, T.PROCESS_ID, T.EQP_TYPE, T.PROCESS_EQP_ID, T.CHAMBER_ID, T.DELAY_TIME, T.REWORK_MAIN_STEP, T.REWORK_YN, T.LAYER_ID, T.STEP_SEQ, T.LOT_TRANSN_DESC, T.PPID, T.TKIN_OPERTR_ID, T.LINE_ID, T.CURRENT_LINE_ID, T.RECENT_TKOUT_DATE, T.TKIN_DATE, T.TKOUT_DATE, T.NET, T.WAIT, MC.START_QTY, MC.FO매수, MC.START_DATE, MC.LAST_EVENT_DATE, MC.TAT, MC.SPLIT_DATE, MC.ID구분1, MC.ID구분2, MC.LOT_LEVEL, T.PORT_LOAD_DATE, T.PORT_UNLOAD_DATE, T.PROCESS_START_DATE, T.PROCESS_FINISH_DATE
FROM MC, T
WHERE T.LOT_ID=MC.LOT_ID(+)

ORDER BY REV_LOT_ID, TKIN_DATE
